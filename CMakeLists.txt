cmake_minimum_required(VERSION 3.21)
project(DbSchemaStudio VERSION 0.9.7 LANGUAGES CXX)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets PrintSupport)
add_executable(DbSchemaStudio WIN32 main.cpp model.cpp logical.cpp rise_import.cpp physical.cpp assets/icons.qrc)
if(WIN32)
 enable_language(RC)
 target_sources(DbSchemaStudio PRIVATE app.rc)
endif()
add_custom_command(TARGET DbSchemaStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/translations" "$<TARGET_FILE_DIR:DbSchemaStudio>/translations")
target_link_libraries(DbSchemaStudio PRIVATE Qt6::Widgets Qt6::PrintSupport)
if(MSVC)
 target_compile_options(DbSchemaStudio PRIVATE /utf-8 /W4)
else()
 target_compile_options(DbSchemaStudio PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_custom_command(TARGET DbSchemaStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/fixtures" "$<TARGET_FILE_DIR:DbSchemaStudio>/fixtures")

foreach(FIXTURE_DIR esempi/Ordini-guida esempi/Esercizi-guidati tests/fixtures/test-limite-azienda)
 add_custom_command(TARGET DbSchemaStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/${FIXTURE_DIR}" "$<TARGET_FILE_DIR:DbSchemaStudio>/test-fixtures")
endforeach()

add_custom_command(TARGET DbSchemaStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/esempi" "$<TARGET_FILE_DIR:DbSchemaStudio>/esempi")
